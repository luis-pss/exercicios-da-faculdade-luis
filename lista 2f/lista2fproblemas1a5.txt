problema 1:

se X > Y e Z != 0 e X + Z > 50 ou Y % 2 == 0 e >= 0:
	print ("condicao atendida")
senão:
	print ("condicao nao atendida")

problema 2:

	print ("valores aceitos")
senão:
	print ("valores nao aceitos")

problema 3:

se A >= 0 e B > A ou C % 2 == 0 e C > 10 e (A + B + C) > 100:
	print ("Aprovado.")
senão:
	print ("Reprovado.")
